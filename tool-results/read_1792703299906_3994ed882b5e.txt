     1→// GigSure AI — Prisma schema
     2→// Work with confidence. Get paid with trust.
     3→//
     4→// Design notes:
     5→// - Database: SQLite (local MVP). Production should swap to PostgreSQL — see DEPLOYMENT_GUIDE.md.
     6→// - Financial values are stored as Integer minor units (e.g., NGN kobo) to avoid binary
     7→//   floating-point rounding errors. Currency codes are stored as ISO 4217 strings.
     8→// - Enums are stored as String with server-side validation (SQLite has no native ENUM).
     9→// - All timestamps are UTC (DateTime).
    10→// - Soft-delete columns (`deletedAt`) are used where data must be retained for legal/audit reasons.
    11→
    12→generator client {
    13→  provider = "prisma-client-js"
    14→}
    15→
    16→datasource db {
    17→  provider = "postgresql"
    18→  url      = env("DATABASE_URL")
    19→}

   456→// PROJECT EXPENSES — service provider requests funds from already-funded milestone
   457→// ---------------------------------------------------------------------------

   459→model ProjectExpense {
   460→  // Service provider requests a partial release of milestone funds for
   461→  // project-related expenses (domain, hosting, software licenses, etc.).
   462→  // The milestone must already be FUNDED — the money is already held in
   463→  // GigSure Payment Protection. The expense request is a request to release
   464→  // a portion of those funds to the service provider before the milestone
   465→  // is completed.
   466→  id              String   @id @default(cuid())
   467→  projectId       String
   468→  milestoneId     String
   469→  requestedById   String   // service provider who needs the money
   470→  approvedById    String?  // client who approved (null until approved)
   471→
   472→  title           String   // e.g., "Domain and hosting"
   473→  description     String?  // breakdown of what the money is for
   474→  category        String   // DOMAIN_HOSTING | SOFTWARE_LICENSE | STOCK_ASSETS | THIRD_PARTY_SERVICES | HARDWARE | OTHER
   475→  amountMinor     Int      // amount requested
   476→  currency        String   @default("NGN")
   477→
   478→  // State machine: REQUESTED → APPROVED → RELEASED | REJECTED | DISPUTED
   479→  status          String   @default("REQUESTED")
   480→  rejectionReason String?
   481→  receiptUrl      String?  // uploaded by service provider after spending (proof of purchase)
   482→  receiptUploadedAt DateTime?
   483→
   484→  // Links to the funding record created when the expense is released
   485→  fundingRecordId String?  @unique
   486→  payoutId        String?  // link to the Payout record when money is sent
   487→
   488→  createdAt       DateTime @default(now())
   489→  approvedAt      DateTime?
   490→  releasedAt      DateTime?
   491→  updatedAt       DateTime @updatedAt
   492→
   493→  project         Project          @relation(fields: [projectId], references: [id], onDelete: Cascade)
   494→  milestone       Milestone        @relation(fields: [milestoneId], references: [id])
   495→  requestedBy     User             @relation("ExpenseRequester", fields: [requestedById], references: [id])
   496→  approvedBy      User?            @relation("ExpenseApprover", fields: [approvedById], references: [id])
   497→  fundingRecord   FundingRecord?   @relation(fields: [fundingRecordId], references: [id])
   498→  auditEvents     AuditEvent[]
   499→
   500→  @@index([projectId])
   501→  @@index([milestoneId])
   502→  @@index([requestedById])
   503→  @@index([status])
   504→}
