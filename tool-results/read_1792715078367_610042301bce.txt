     1→// GigSure AI — Prisma schema
     2→// Work with confidence. Get paid with trust.
     3→//
     4→// Design notes:
     5→// - Database: SQLite (local MVP). Production should swap to PostgreSQL — see DEPLOYMENT_GUIDE.md.
     6→// - Financial values are stored as Integer minor units (e.g., NGN kobo) to avoid binary
     7→//   floating-point rounding errors. Currency codes are stored as ISO 4217 strings.
     8→// - Enums are stored as String with server-side validation (SQLite has no native ENUM).
     9→// - All timestamps are UTC (DateTime).
    10→// - Soft-delete columns (`deletedAt`) are used where data must be retained for legal/audit reasons.
    11→
    12→generator client {
    13→  provider = "prisma-client-js"
    14→}
    15→
    16→datasource db {
    17→  provider = "postgresql"
    18→  url      = env("DATABASE_URL")
    19→}

   464→model ProjectExpense {
   465→  // Service provider requests a partial release of milestone funds for
   466→  // project-related expenses (domain, hosting, software licenses, etc.).
   467→  // The milestone must already be FUNDED — the money is already held in
   468→  // GigSure Payment Protection. The expense request is a request to release
   469→  // a portion of those funds to the service provider before the milestone
   470→  // is completed.
   471→  id              String   @id @default(cuid())
   472→  projectId       String
   473→  milestoneId     String
   474→  requestedById   String   // service provider who needs the money
   475→  approvedById    String?  // client who approved (null until approved)
   476→
   477→  title           String   // e.g., "Domain and hosting"
   478→  description     String?  // breakdown of what the money is for
   479→  category        String   // DOMAIN_HOSTING | SOFTWARE_LICENSE | STOCK_ASSETS | THIRD_PARTY_SERVICES | HARDWARE | OTHER
   480→  amountMinor     Int      // amount requested
   481→  currency        String   @default("NGN")
   482→
   483→  // State machine: REQUESTED → APPROVED → RELEASED | REJECTED | DISPUTED
   484→  status          String   @default("REQUESTED")
   485→  rejectionReason String?
   486→  receiptUrl      String?  // uploaded by service provider after spending (proof of purchase)
   487→  receiptUploadedAt DateTime?
   488→
   489→  // Links to the funding record created when the expense is released
   490→  fundingRecordId String?  @unique
   491→  payoutId        String?  // link to the Payout record when money is sent
   492→
   493→  createdAt       DateTime @default(now())
   494→  approvedAt      DateTime?
   495→  releasedAt      DateTime?
   496→  updatedAt       DateTime @updatedAt
   497→
   498→  project         Project          @relation(fields: [projectId], references: [id], onDelete: Cascade)
   499→  milestone       Milestone        @relation(fields: [milestoneId], references: [id])
   500→  requestedBy     User             @relation("ExpenseRequester", fields: [requestedById], references: [id])
   501→  approvedBy      User?            @relation("ExpenseApprover", fields: [approvedById], references: [id])
   502→  fundingRecord   FundingRecord?   @relation(fields: [fundingRecordId], references: [id])
   503→  auditEvents     AuditEvent[]
   504→
   505→  @@index([projectId])
   506→  @@index([milestoneId])
   507→  @@index([requestedById])
   508→  @@index([status])
   509→}
