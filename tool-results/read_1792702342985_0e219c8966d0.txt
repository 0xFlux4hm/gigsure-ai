     1→// GigSure AI — Prisma schema
     2→// Work with confidence. Get paid with trust.
     3→//
     4→// Design notes:
     5→// - Database: SQLite (local MVP). Production should swap to PostgreSQL — see DEPLOYMENT_GUIDE.md.
     6→// - Financial values are stored as Integer minor units (e.g., NGN kobo) to avoid binary
     7→//   floating-point rounding errors. Currency codes are stored as ISO 4217 strings.
     8→// - Enums are stored as String with server-side validation (SQLite has no native ENUM).
     9→// - All timestamps are UTC (DateTime).
    10→// - Soft-delete columns (`deletedAt`) are used where data must be retained for legal/audit reasons.
    11→
    12→generator client {
    13→  provider = "prisma-client-js"
    14→}
    15→
    16→datasource db {
    17→  provider = "postgresql"
    18→  url      = env("DATABASE_URL")
    19→}

   456→model ProjectExpense {
   457→  // Service provider requests a partial release of milestone funds for
   458→  // project-related expenses (domain, hosting, software licenses, etc.).
   459→  // The milestone must already be FUNDED — the money is already held in
   460→  // GigSure Payment Protection. The expense request is a request to release
   461→  // a portion of those funds to the service provider before the milestone
   462→  // is completed.
   463→  id              String   @id @default(cuid())
   464→  projectId       String
   465→  milestoneId     String
   466→  requestedById   String   // service provider who needs the money
   467→  approvedById    String?  // client who approved (null until approved)
   468→
   469→  title           String   // e.g., "Domain and hosting"
   470→  description     String?  // breakdown of what the money is for
   471→  category        String   // DOMAIN_HOSTING | SOFTWARE_LICENSE | STOCK_ASSETS | THIRD_PARTY_SERVICES | HARDWARE | OTHER
   472→  amountMinor     Int      // amount requested
   473→  currency        String   @default("NGN")
   474→
   475→  // State machine: REQUESTED → APPROVED → RELEASED | REJECTED | DISPUTED
   476→  status          String   @default("REQUESTED")
   477→  rejectionReason String?
   478→  receiptUrl      String?  // uploaded by service provider after spending (proof of purchase)
   479→  receiptUploadedAt DateTime?
   480→
   481→  // Links to the funding record created when the expense is released
   482→  fundingRecordId String?  @unique
   483→  payoutId        String?  // link to the Payout record when money is sent
   484→
   485→  createdAt       DateTime @default(now())
   486→  approvedAt      DateTime?
   487→  releasedAt      DateTime?
   488→  updatedAt       DateTime @updatedAt
   489→
   490→  project         Project          @relation(fields: [projectId], references: [id], onDelete: Cascade)
   491→  milestone       Milestone        @relation(fields: [milestoneId], references: [id])
   492→  requestedBy     User             @relation("ExpenseRequester", fields: [requestedById], references: [id])
   493→  approvedBy      User?            @relation("ExpenseApprover", fields: [approvedById], references: [id])
   494→  fundingRecord   FundingRecord?   @relation(fields: [fundingRecordId], references: [id])
   495→  auditEvents     AuditEvent[]
   496→
   497→  @@index([projectId])
   498→  @@index([milestoneId])
   499→  @@index([requestedById])
   500→  @@index([status])
   501→}
