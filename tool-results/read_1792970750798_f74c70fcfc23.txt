    15→/**
    16→ * Check if Google OAuth is configured by fetching the auth providers list.
    17→ * Returns true only if the "google" provider is registered.
    18→ */
    19→function useGoogleAuthAvailable(): boolean {
    20→  const [available, setAvailable] = useState(false);
    21→  useEffect(() => {
    22→    api.get<Record<string, { id: string }>>("/api/auth/providers")
    23→      .then((providers) => {
    24→        setAvailable(!!providers?.google);
    25→      })
    26→      .catch(() => setAvailable(false));
    27→  }, []);
    28→  return available;
    29→}

   294→function RegisterForm() {
   295→  const { navigate } = useNav();
   296→  const [email, setEmail] = useState("");
   297→  const [displayName, setDisplayName] = useState("");
   298→  const [password, setPassword] = useState("");
   299→  const [terms, setTerms] = useState(false);
   300→  const [loading, setLoading] = useState(false);
   301→  const [error, setError] = useState<string | null>(null);
   302→  const [verifyUrl, setVerifyUrl] = useState<string | null>(null);
   303→
   304→  const submit = async (e: React.FormEvent) => {
   305→    e.preventDefault();
   306→    setError(null);
   307→    if (!terms) {
   308→      setError("Please accept the terms to continue.");
   309→      return;
   310→    }
   311→    setLoading(true);
   312→    try {
   313→      const res = await api.post<{ verificationUrl?: string; message?: string; created?: boolean; emailSent?: boolean; emailError?: string }>("/api/auth/register", {
   314→        email,
   315→        displayName,
   316→        password,
   317→        termsAccepted: true,
   318→      });
   319→      if (res.verificationUrl) setVerifyUrl(res.verificationUrl);
   320→      if (res.created) {
   321→        // Auto-login after registration, then redirect to dashboard
   322→        const signInResult = await signIn("credentials", {
   323→          email,
   324→          password,
   325→          redirect: false,
   326→          callbackUrl: "/?view=dashboard",
   327→        });
   328→        if (signInResult?.ok) {
   329→          // Redirect to dashboard on successful auto-login
   330→          window.location.href = signInResult.url ?? "/?view=dashboard";
   331→        } else {
   332→          // Auto-login failed — email verification is required before login.
   333→          // Show a clear message about whether the email was actually sent.
   334→          if (res.emailSent === false) {
   335→            setError(
   336→              "Account created, but the verification email could not be sent. " +
   337→              "Please contact support."
   338→            );
   339→          } else {
   340→            setError(
   341→              "Account created! We sent a verification link to your email. " +
   342→              "Click the link in the email to verify your account, then sign in."
   343→            );
   344→          }
   345→        }
   346→      }
   347→    } catch (err) {
   348→      setError((err as Error).message ?? "Registration failed");
   349→    } finally {
   350→      setLoading(false);
   351→    }
   352→  };
   353→
   354→  return (
   355→    <Card>
   356→      <CardHeader>
   357→        <CardTitle>Create your account</CardTitle>
   358→        <CardDescription>
   359→          Free for individuals. No credit card required.
   360→        </CardDescription>
   361→      </CardHeader>
   362→      <CardContent className="space-y-4">
   363→        <GoogleButton label="Sign up with Google" />
   364→        <GoogleDivider />
   365→        <form onSubmit={submit} className="space-y-3">
   366→          <div className="space-y-1.5">
   367→            <Label htmlFor="displayName">Display name</Label>
   368→            <Input
   369→              id="displayName"
   370→              type="text"
   371→              value={displayName}
   372→              onChange={(e) => setDisplayName(e.target.value)}
   373→              required
   374→              disabled={loading}
   375→              maxLength={80}
   376→            />
   377→          </div>
   378→          <div className="space-y-1.5">
   379→            <Label htmlFor="email">Email</Label>
   380→            <Input
   381→              id="email"
   382→              type="email"
   383→              autoComplete="email"
   384→              value={email}
   385→              onChange={(e) => setEmail(e.target.value)}
   386→              required
   387→              disabled={loading}
   388→            />
   389→          </div>
   390→          <div className="space-y-1.5">
   391→            <Label htmlFor="password">Password</Label>
   392→            <Input
   393→              id="password"
   394→              type="password"
   395→              autoComplete="new-password"
   396→              value={password}
   397→              onChange={(e) => setPassword(e.target.value)}
   398→              required
   399→              disabled={loading}
   400→              minLength={8}
   401→            />
   402→            <p className="text-xs text-muted-foreground">At least 8 characters, with a letter and a digit.</p>
   403→          </div>
   404→          <label className="flex items-start gap-2 text-sm">
   405→            <input
   406→              type="checkbox"
   407→              checked={terms}
   408→              onChange={(e) => setTerms(e.target.checked)}
   409→              className="mt-1"
   410→            />
   411→            <span className="text-muted-foreground">
   412→              I accept the Terms of Service and Privacy Notice. I understand
   413→              this is a project collaboration platform, not a payment custodian.
   414→            </span>
   415→          </label>
   416→          {error && (
   417→            <Alert variant="destructive">
   418→              <AlertDescription>{error}</AlertDescription>
   419→            </Alert>
   420→          )}
   421→          <Button type="submit" className="w-full" disabled={loading}>
   422→            {loading ? <Loader2 className="h-4 w-4 animate-spin mr-2" /> : null}
   423→            Create account
   424→          </Button>
   425→        </form>
   426→        {verifyUrl && (
   427→          <Alert>
   428→            <AlertDescription>
   429→              Verification link generated (dev mode):{" "}
   430→              <a href={verifyUrl} className="text-primary hover:underline break-all">
   431→                Click to verify your email
   432→              </a>
   433→            </AlertDescription>
   434→          </Alert>
   435→        )}
   436→      </CardContent>
   437→      <CardFooter className="flex-col gap-2 text-sm text-muted-foreground">
   438→        <p>Already have an account?</p>
   439→        <Button variant="outline" className="w-full" onClick={() => navigate({ view: "login" })}>
   440→          Sign in
   441→        </Button>
   442→      </CardFooter>
   443→    </Card>
   444→  );

   447→function VerifyEmailForm({ token }: { token?: string }) {
   448→  const { navigate } = useNav();
   449→  const [enteredToken, setEnteredToken] = useState(token ?? "");
   450→  const [loading, setLoading] = useState(false);
   451→  const [ok, setOk] = useState(false);
   452→  const [error, setError] = useState<string | null>(null);
   453→
   454→  const submit = async (e: React.FormEvent) => {
   455→    e.preventDefault();
   456→    setLoading(true);
   457→    setError(null);
   458→    try {
   459→      await api.post("/api/auth/verify-email", { token: enteredToken });
   460→      setOk(true);
   461→      setTimeout(() => navigate({ view: "login" }), 2000);
   462→    } catch (err) {
   463→      setError((err as Error).message ?? "Verification failed");
   464→    } finally {
   465→      setLoading(false);
   466→    }
   467→  };
   468→
   469→  return (
   470→    <Card>
   471→      <CardHeader>
   472→        <CardTitle>Verify your email</CardTitle>
   473→        <CardDescription>
   474→          Enter the verification token emailed to you. In dev mode, the link is shown after registration.
   475→        </CardDescription>
   476→      </CardHeader>
   477→      <CardContent>
   478→        <form onSubmit={submit} className="space-y-3">
   479→          <div className="space-y-1.5">
   480→            <Label htmlFor="token">Verification token</Label>
   481→            <Input
   482→              id="token"
   483→              type="text"
   484→              value={enteredToken}
   485→              onChange={(e) => setEnteredToken(e.target.value)}
   486→              required
   487→              disabled={loading}
   488→            />
   489→          </div>
   490→          {error && (
   491→            <Alert variant="destructive">
   492→              <AlertDescription>{error}</AlertDescription>
   493→            </Alert>
   494→          )}
   495→          {ok && (
   496→            <Alert>
   497→              <AlertDescription>Email verified. Redirecting to sign-in…</AlertDescription>
   498→            </Alert>
   499→          )}
   500→          <Button type="submit" className="w-full" disabled={loading}>
   501→            {loading ? <Loader2 className="h-4 w-4 animate-spin mr-2" /> : null}
   502→            Verify email
   503→          </Button>
   504→        </form>
   505→      </CardContent>
   506→    </Card>
   507→  );
   508→}

   651→function ResetPasswordForm({ token }: { token?: string }) {
   652→  const { navigate } = useNav();
   653→  const [enteredToken, setEnteredToken] = useState(token ?? "");
   654→  const [newPassword, setNewPassword] = useState("");
   655→  const [loading, setLoading] = useState(false);
   656→  const [ok, setOk] = useState(false);
   657→  const [error, setError] = useState<string | null>(null);
   658→
   659→  const submit = async (e: React.FormEvent) => {
   660→    e.preventDefault();
   661→    setLoading(true);
   662→    setError(null);
   663→    try {
   664→      await api.post("/api/auth/reset-password", { token: enteredToken, newPassword });
   665→      setOk(true);
   666→      setTimeout(() => navigate({ view: "login" }), 2000);
   667→    } catch (err) {
   668→      setError((err as Error).message ?? "Reset failed");
   669→    } finally {
   670→      setLoading(false);
   671→    }
   672→  };
   673→
   674→  return (
   675→    <Card>
   676→      <CardHeader>
   677→        <CardTitle>Set a new password</CardTitle>
   678→        <CardDescription>Use a strong password you haven&apos;t used before.</CardDescription>
   679→      </CardHeader>
   680→      <CardContent>
   681→        {ok ? (
   682→          <Alert>
   683→            <AlertDescription>Password reset. Redirecting to sign-in…</AlertDescription>
   684→          </Alert>
   685→        ) : (
   686→          <form onSubmit={submit} className="space-y-3">
   687→            <div className="space-y-1.5">
   688→              <Label htmlFor="token">Reset token</Label>
   689→              <Input
   690→                id="token"
   691→                type="text"
   692→                value={enteredToken}
   693→                onChange={(e) => setEnteredToken(e.target.value)}
   694→                required
   695→                disabled={loading}
   696→              />
   697→            </div>
   698→            <div className="space-y-1.5">
   699→              <Label htmlFor="newpassword">New password</Label>
   700→              <Input
   701→                id="newpassword"
   702→                type="password"
   703→                value={newPassword}
   704→                onChange={(e) => setNewPassword(e.target.value)}
   705→                required
   706→                disabled={loading}
   707→                minLength={8}
   708→              />
   709→            </div>
   710→            {error && (
   711→              <Alert variant="destructive">
   712→                <AlertDescription>{error}</AlertDescription>
   713→              </Alert>
   714→            )}
   715→            <Button type="submit" className="w-full" disabled={loading}>
   716→              {loading ? <Loader2 className="h-4 w-4 animate-spin mr-2" /> : null}
   717→              Reset password
   718→            </Button>
   719→          </form>
   720→        )}
   721→      </CardContent>
   722→    </Card>
   723→  );
   724→}
