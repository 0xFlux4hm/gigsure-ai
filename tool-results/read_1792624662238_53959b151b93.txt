   162→    setError(null);

   164→  };

   167→  if (loading) return <LoadingState />;

   237→function StatCard({ label, value, sub, icon: Icon, accent = "default" }: {
   238→  label: string;
   239→  value: number;
   240→  sub: string;
   241→  icon: React.ComponentType<{ className?: string }>;
   242→  accent?: "default" | "primary" | "destructive";
   243→}) {
   244→  const accentClasses = {
   245→    default: "bg-muted/30",
   246→    primary: "bg-primary/10 text-primary",
   247→    destructive: "bg-destructive/10 text-destructive",
   248→  };
   249→  return (
   250→    <Card>
   251→      <CardContent className="p-4">
   252→        <div className="flex items-start justify-between gap-2">
   253→          <div className="min-w-0">
   254→            <div className="text-xs text-muted-foreground">{label}</div>
   255→            <div className="text-2xl font-bold mt-1">{value}</div>
   256→            <div className="text-xs text-muted-foreground mt-1">{sub}</div>
   257→          </div>
   258→          <div className={`h-9 w-9 rounded-md flex items-center justify-center shrink-0 ${accentClasses[accent]}`}>
   259→            <Icon className="h-5 w-5" />
   260→          </div>
   261→        </div>
   262→      </CardContent>
   263→    </Card>
   264→  );
   265→}

   291→type AdminUser = {
   292→  id: string;
   293→  email: string;
   294→  displayName: string;
   295→  emailVerifiedAt: string | null;
   296→  accountStatus: string;
   297→  platformRole: string;
   298→  lastLoginAt: string | null;

   300→  deletedAt: string | null;
   301→  _count: { projectsOwned: number; memberships: number; auditEvents: number; aiActivities: number };

   303→
   304→function UsersTab() {
   305→  const [users, setUsers] = useState<AdminUser[]>([]);
   306→  const [total, setTotal] = useState(0);
   307→  const [loading, setLoading] = useState(true);
   308→  const [search, setSearch] = useState("");
   309→  const [statusFilter, setStatusFilter] = useState<string>("");
   310→  const [roleFilter, setRoleFilter] = useState<string>("");
   311→  const [actingId, setActingId] = useState<string | null>(null);
   312→
   313→  const load = () => {
   314→    setLoading(true);
   315→    const params = new URLSearchParams();
   316→    if (search) params.set("q", search);
   317→    if (statusFilter) params.set("status", statusFilter);
   318→    if (roleFilter) params.set("role", roleFilter);
   319→    api.get<{ users: AdminUser[]; total: number }>(`/api/admin/users?${params.toString()}`)
   320→      .then((r) => { setUsers(r.users); setTotal(r.total); })
   321→      .finally(() => setLoading(false));
   322→  };
   323→  useEffect(load, [search, statusFilter, roleFilter]);
   324→
   325→  const act = async (userId: string, action: "suspend" | "activate" | "promote_admin" | "demote_admin") => {
   326→    if (!confirm(`Are you sure you want to ${action.replace("_", " ")} this user?`)) return;
   327→    setActingId(userId);
   328→    try {
   329→      const reason = action === "suspend" ? prompt("Reason for suspension (optional):") ?? "" : "";
   330→      await api.patch(`/api/admin/users/${userId}`, { action, reason: reason || undefined });
   331→      load();
   332→    } catch (err) {
   333→      alert((err as Error).message);
   334→    } finally {
   335→      setActingId(null);
   336→    }
   337→  };
   338→
   339→  return (
   340→    <div className="space-y-4">
   341→      <Card>
   342→        <CardContent className="p-3 flex flex-wrap gap-2 items-end">
   343→          <div className="flex-1 min-w-[200px] space-y-1">
   344→            <Label className="text-xs">Search</Label>
   345→            <div className="relative">
   346→              <Search className="absolute left-2 top-2.5 h-4 w-4 text-muted-foreground" />
   347→              <Input
   348→                className="pl-8"
   349→                placeholder="Email or name…"
   350→                value={search}
   351→                onChange={(e) => setSearch(e.target.value)}
   352→              />
   353→            </div>
   354→          </div>
   355→          <div className="space-y-1">
   356→            <Label className="text-xs">Status</Label>
   357→            <Select value={statusFilter || "ALL"} onValueChange={(v) => setStatusFilter(v === "ALL" ? "" : v)}>
   358→              <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   359→              <SelectContent>
   360→                <SelectItem value="ALL">All</SelectItem>
   361→                <SelectItem value="ACTIVE">Active</SelectItem>
   362→                <SelectItem value="SUSPENDED">Suspended</SelectItem>
   363→                <SelectItem value="PENDING_DELETION">Pending deletion</SelectItem>
   364→                <SelectItem value="DELETED">Deleted</SelectItem>
   365→              </SelectContent>
   366→            </Select>
   367→          </div>
   368→          <div className="space-y-1">
   369→            <Label className="text-xs">Role</Label>
   370→            <Select value={roleFilter || "ALL"} onValueChange={(v) => setRoleFilter(v === "ALL" ? "" : v)}>
   371→              <SelectTrigger className="w-32"><SelectValue /></SelectTrigger>
   372→              <SelectContent>
   373→                <SelectItem value="ALL">All</SelectItem>
   374→                <SelectItem value="USER">User</SelectItem>
   375→                <SelectItem value="ADMIN">Admin</SelectItem>
   376→              </SelectContent>
   377→            </Select>
   378→          </div>
   379→        </CardContent>
   380→      </Card>
   381→
   382→      <Card>
   383→        <CardHeader className="pb-3">
   384→          <CardTitle className="text-base flex items-center justify-between">
   385→            <span>Users ({total})</span>
   386→          </CardTitle>
   387→        </CardHeader>
   388→        <CardContent className="p-0">
   389→          {loading ? (
   390→            <LoadingState />
   391→          ) : (
   392→            <div className="overflow-x-auto">
   393→              <table className="w-full text-sm">
   394→                <thead className="bg-muted/30 text-xs text-muted-foreground">
   395→                  <tr>
   396→                    <th className="text-left p-2">User</th>
   397→                    <th className="text-left p-2">Status</th>
   398→                    <th className="text-left p-2">Role</th>
   399→                    <th className="text-right p-2">Projects</th>
   400→                    <th className="text-right p-2">AI calls</th>
   401→                    <th className="text-left p-2">Last login</th>
   402→                    <th className="text-right p-2">Actions</th>
   403→                  </tr>
   404→                </thead>
   405→                <tbody>
   406→                  {users.length === 0 && (
   407→                    <tr><td colSpan={7} className="p-6 text-center text-muted-foreground">No users found.</td></tr>
   408→                  )}
   409→                  {users.map((u) => (
   410→                    <tr key={u.id} className="border-t hover:bg-muted/30">
   411→                      <td className="p-2">
   412→                        <div className="font-medium">{u.displayName}</div>
   413→                        <div className="text-xs text-muted-foreground">{u.email}</div>
   414→                      </td>
   415→                      <td className="p-2">
   416→                        <Badge variant={
   417→                          u.accountStatus === "ACTIVE" ? "default" :
   418→                          u.accountStatus === "SUSPENDED" ? "destructive" :
   419→                          u.accountStatus === "DELETED" ? "secondary" :
   420→                          "outline"
   421→                        } className="text-xs">{u.accountStatus}</Badge>
   422→                      </td>
   423→                      <td className="p-2">
   424→                        <Badge variant={u.platformRole === "ADMIN" ? "default" : "outline"} className="text-xs">{u.platformRole}</Badge>
   425→                      </td>
   426→                      <td className="p-2 text-right">{u._count.projectsOwned}</td>
   427→                      <td className="p-2 text-right">{u._count.aiActivities}</td>
   428→                      <td className="p-2 text-xs text-muted-foreground">{u.lastLoginAt ? timeAgo(u.lastLoginAt) : "never"}</td>
   429→                      <td className="p-2 text-right">
   430→                        {actingId === u.id ? (
   431→                          <Loader2 className="h-4 w-4 animate-spin inline" />
   432→                        ) : (
   433→                          <div className="flex justify-end gap-1">
   434→                            {u.accountStatus === "ACTIVE" ? (
   435→                              <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => act(u.id, "suspend")}>Suspend</Button>
   436→                            ) : u.accountStatus === "SUSPENDED" ? (
   437→                              <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => act(u.id, "activate")}>Activate</Button>
   438→                            ) : null}
   439→                            {u.platformRole === "USER" ? (
   440→                              <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => act(u.id, "promote_admin")}>Promote</Button>
   441→                            ) : (
   442→                              <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => act(u.id, "demote_admin")}>Demote</Button>
   443→                            )}
   444→                          </div>
   445→                        )}
   446→                      </td>
   447→                    </tr>
   448→                  ))}
   449→                </tbody>
   450→              </table>
   451→            </div>
   452→          )}
   453→        </CardContent>
   454→      </Card>
   455→    </div>
