     1→// GigSure AI — Prisma schema
     2→// Work with confidence. Get paid with trust.
     3→//
     4→// Design notes:
     5→// - Database: SQLite (local MVP). Production should swap to PostgreSQL — see DEPLOYMENT_GUIDE.md.
     6→// - Financial values are stored as Integer minor units (e.g., NGN kobo) to avoid binary
     7→//   floating-point rounding errors. Currency codes are stored as ISO 4217 strings.
     8→// - Enums are stored as String with server-side validation (SQLite has no native ENUM).
     9→// - All timestamps are UTC (DateTime).
    10→// - Soft-delete columns (`deletedAt`) are used where data must be retained for legal/audit reasons.
    11→
    12→generator client {
    13→  provider = "prisma-client-js"
    14→}
    15→
    16→datasource db {
    17→  provider = "postgresql"
    18→  url      = env("DATABASE_URL")
    19→}

   460→model ProjectExpense {
   461→  // Service provider requests a partial release of milestone funds for
   462→  // project-related expenses (domain, hosting, software licenses, etc.).
   463→  // The milestone must already be FUNDED — the money is already held in
   464→  // GigSure Payment Protection. The expense request is a request to release
   465→  // a portion of those funds to the service provider before the milestone
   466→  // is completed.
   467→  id              String   @id @default(cuid())
   468→  projectId       String
   469→  milestoneId     String
   470→  requestedById   String   // service provider who needs the money
   471→  approvedById    String?  // client who approved (null until approved)
   472→
   473→  title           String   // e.g., "Domain and hosting"
   474→  description     String?  // breakdown of what the money is for
   475→  category        String   // DOMAIN_HOSTING | SOFTWARE_LICENSE | STOCK_ASSETS | THIRD_PARTY_SERVICES | HARDWARE | OTHER
   476→  amountMinor     Int      // amount requested
   477→  currency        String   @default("NGN")
   478→
   479→  // State machine: REQUESTED → APPROVED → RELEASED | REJECTED | DISPUTED
   480→  status          String   @default("REQUESTED")
   481→  rejectionReason String?
   482→  receiptUrl      String?  // uploaded by service provider after spending (proof of purchase)
   483→  receiptUploadedAt DateTime?
   484→
   485→  // Links to the funding record created when the expense is released
   486→  fundingRecordId String?  @unique
   487→  payoutId        String?  // link to the Payout record when money is sent
   488→
   489→  createdAt       DateTime @default(now())
   490→  approvedAt      DateTime?
   491→  releasedAt      DateTime?
   492→  updatedAt       DateTime @updatedAt
   493→
   494→  project         Project          @relation(fields: [projectId], references: [id], onDelete: Cascade)
   495→  milestone       Milestone        @relation(fields: [milestoneId], references: [id])
   496→  requestedBy     User             @relation("ExpenseRequester", fields: [requestedById], references: [id])
   497→  approvedBy      User?            @relation("ExpenseApprover", fields: [approvedById], references: [id])
   498→  fundingRecord   FundingRecord?   @relation(fields: [fundingRecordId], references: [id])
   499→  auditEvents     AuditEvent[]
   500→
   501→  @@index([projectId])
   502→  @@index([milestoneId])
   503→  @@index([requestedById])
   504→  @@index([status])
   505→}
