    15→    15→/**
    16→    16→ * Check if Google OAuth is configured by fetching the auth providers list.
    17→    17→ * Returns true only if the "google" provider is registered.
    18→    18→ */
    19→    19→function useGoogleAuthAvailable(): boolean {
    20→    20→  const [available, setAvailable] = useState(false);
    21→    21→  useEffect(() => {
    22→    22→    api.get<Record<string, { id: string }>>("/api/auth/providers")
    23→    23→      .then((providers) => {
    24→    24→        setAvailable(!!providers?.google);
    25→    25→      })
    26→    26→      .catch(() => setAvailable(false));
    27→    27→  }, []);
    28→    28→  return available;
    29→    29→}
